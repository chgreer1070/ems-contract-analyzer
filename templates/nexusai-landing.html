// Connect, explore, deploy

import { NexusAI } from '@nexusai/sdk';

const platform = new NexusAI({
  sources: ['snowflake', 's3://docs'],
  mode: 'auto-discover'
});

// Unified query across all sources
const insights = await platform.analyze({
  query: 'root cause of Q3 churn',
  explain: true,
  confidence: 0.95
});

// Deploy to production
await platform.deploy(insights, {
  schedule: 'realtime',
  alerts: true
});

// → 12 sources unified
// → Root cause identified: 94.7%
// → 3 actionable recommendations
